% Figure placeholders, per figure: an \includegraphics whose file is in
% graphic/ renders normally; one whose file is not there yet draws a grey
% box of the requested size labelled with the missing filename, instead
% of stopping the compile with "File not found". So you can add images
% one at a time (tick them off in FIGURE_SOURCES.md) and recompile.
\makeatletter
\AtBeginDocument{%
  \let\Gin@include@real\Ginclude@graphics
  \def\Ginclude@graphics#1{%
    \begingroup
      \let\input@path\Ginput@path
      \IfFileExists{#1}{\endgroup\Gin@include@real{#1}}%
        {\endgroup\Gin@placeholder{#1}}}%
}
\def\Gin@placeholder#1{%
  \begingroup
    \@ifundefined{Gin@@ewidth}{\dimen@=150pt}{\dimen@=\Gin@@ewidth\relax}%
    \@ifundefined{Gin@@eheight}{\dimen2=0.4\dimen@}{\dimen2=\Gin@@eheight\relax}%
    \setlength{\fboxsep}{0pt}%
    \fbox{\colorbox{black!8}{\parbox[c][\dimen2][c]{\dimen@}{\centering
      \footnotesize\textsf{image not added yet:}\\[2pt]
      \texttt{\detokenize{#1}}}}}%
  \endgroup}
\makeatother
